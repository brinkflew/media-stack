# ==============================================================================
# Pod: torrent
# ------------------------------------------------------------------------------
# The Podman equivalent of Compose's `network_mode: "service:gluetun"`. gluetun,
# qBittorrent and JOAL share one network namespace, so if the VPN drops they
# lose connectivity entirely - a kill-switch by construction rather than by a
# firewall rule someone has to remember.
#
# THE POD OWNS THE NETWORK. Its members inherit it and must not declare a
# Network= of their own; doing so would give a downloader a route outside the
# tunnel, which is the one failure this whole arrangement exists to prevent.
#
# The pod's infra container holds the namespace, and that namespace has a
# working default route out through the bridge from the moment it is created -
# before gluetun has built the tunnel. That window is why the members are
# health-gated on gluetun rather than merely ordered after it.
#
# No port is published. qBittorrent's and JOAL's web UIs are reached through
# Caddy, which addresses them as gluetun:<port> because neither has a name of
# its own. Incoming peers arrive through the tunnel on the port ProtonVPN
# forwards, landing directly in this namespace, so publishing 6881 on the host
# would only forward to a port nothing listens on.
# ==============================================================================

[Unit]
Description=Torrent pod (VPN namespace shared by gluetun, qBittorrent and JOAL)

[Pod]
PodName=torrent
Network=net-download.network

# THE POD MUST SURVIVE ITS MEMBERS RESTARTING, and by default it does not.
#
# Quadlet creates the pod with `--exit-policy stop`, which stops the pod as soon
# as its last container exits. Restarting any member is therefore a trap:
# systemd stops the dependants first, the last container goes, the exit policy
# stops the pod, torrent-pod.service sees its infra pid disappear and runs
# `podman pod rm --force` - and then the member cannot come back, because it is
# BindsTo=torrent-pod.service and the pod no longer exists. It fails with
#
#   gluetun.service: Bound to unit torrent-pod.service, but unit isn't active.
#
# leaving gluetun, qBittorrent and JOAL all down. Measured, not theorised: a
# plain `systemctl --user restart gluetun` did exactly this and took the whole
# torrent stack out until the pod was started by hand.
#
# That matters far more now than it used to. podman auto-update restarts the
# unit named in each container's PODMAN_SYSTEMD_UNIT, which for a pod member is
# the MEMBER's unit - so an unattended gluetun update would have hit this at
# midnight and left downloads dead until someone noticed.
#
# `continue` keeps the infra container running when members stop, so the
# namespace persists and a restarted member rejoins the pod it was already in.
PodmanArgs=--exit-policy=continue

[Install]
WantedBy=default.target
